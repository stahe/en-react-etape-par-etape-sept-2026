    // =========================================================================
// 01-first-app/src/App.tsx - a first component
// -------------------------------------------------------------------------
// A React component is a FONCTION, whose name begins with an
// uppercase letter, which RENVOIE determines what to display, written in JSX: from HTML
// in the code TypeScript. React calls this function to generate the
// page; it calls RAPPELLE every time the component’s state changes, and uses
// updates the page with the new result.
// =========================================================================

import { useState } from 'react';
// a stylesheet, imported by the component that uses it
import './App.css';

// "ordinary" data, outside the component: it never changes
const titre = 'Bienvenue dans React';
const langages = ['TypeScript', 'HTML', 'CSS'];

export default function App() {
   // useState(0): a data item from ÉTAT, with an initial value of 0. We receive its
   // value ACTUELLE (counter) and the function that changes it (setCompteur).
   // Calling setCompteur instructs React to RAPPELER App to update
   // the page. You should never modify the page yourself.
  const [compteur, setCompteur] = useState(0);
  const [heure, setHeure] = useState(() => new Date().toLocaleTimeString());

   // a function called by the button
  function incrementer() {
    setCompteur((n) => n + 1);
    setHeure(new Date().toLocaleTimeString());
  }

   // the JSX returned: { expression } inserts the value of an expression;
   // className replaces the HTML class attribute (a reserved word in JavaScript)
  return (
    <div className="container py-4">
      <h1 className="h3">{titre}</h1>
      <p>Cette page est fabriquée dans le navigateur, par React, à partir de index.html, main.tsx et du composant App.</p>
      <p>
        On y écrit du {langages.join(', ')} ({langages.length} langages).
      </p>

      {/* onClick: the function to call when clicked (see example 04) */}
      <button className="btn btn-primary" onClick={incrementer}>
        Cliquez-moi
      </button>
      <p className="mt-3 compteur">
        Vous avez cliqué {compteur} fois (dernier clic : {heure}).
      </p>
    </div>
  );
}
    • Line 11: the hook [useState], provided by [React];
    • line 13: the component’s style sheet;
    • lines 16–17: two ordinary data items, declared outside the component: they never change;
    • line 19: the component: a function whose name begins with a capital letter (this is how JSX distinguishes a component, [<App />], from an element, HTML, [<div>]). [export default]: this is what the module exports by default, imported into [main.tsx] by [import App from './App'];
    • Line 24: [useState(0)] creates a state variable with an initial value of 0. It returns an array of two elements, which we “unpack”: the current value ([compteur]) and the function that changes it ([setCompteur]). The state is stored by [React]: it persists from one function call to the next;
    • Line 25: You can assign a function to [useState] that calculates the initial value; it is called only the first time the page is displayed;
    • Lines 28–30: A function called by the button. It does not modify the page; it changes the state. [setCompteur((n) => n + 1)] calculates the new value based on the old one. Each call to the function [setXxx] instructs [React] to call [App]; [React] compares the new result to the old one and updates only the parts of the page that have changed;
    • lines 35–51: the returned JSX, i.e., what should be displayed:
      • the JSX resembles the HTML, with a few differences: the attribute [class] is written as [className] ([class] is a reserved word for [JavaScript]), each tag must be closed ([<br />], [<input />]), and the entire structure must have UN as its root element;
      • Line 37: [{ expression }] inserts the value of an expression [TypeScript]. It is escaped: if [titre] contained [<b>], [<b>] would be displayed without the tag being interpreted;
      • line 40: the expression can be anything: a method call, a calculation;
      • line 43: a comment in JSX is written as [{/* ... */}];
      • line 44: Clicking the button calls the function [incrementer] (see Example 04). The function itself is provided without parentheses: [onClick={incrementer()}] would call it immediately, while it is being displayed;
      • line 48: the current value of the state. When the state changes, [App] is called again, and this text changes.

    JSX is not the same as HTML: it is a convenient notation for function calls. The compiler transforms [<h1 className="h3">{titre}</h1>] into a call of the form [jsx('h1', { className: 'h3', children: titre })], which creates an object describing the element. The component therefore returns a tree of objects (the “virtual DOM”), which [React] compares to the previous tree to determine what to change in the actual page.

    // =========================================================================
// 02-status/src/App.tsx - status, calculated values, useEffect
// -------------------------------------------------------------------------
// A small price calculator TTC:
//    - useStat   e : a value from ÉTAT; changing it (setXxx) triggers
//                    the component, which renders the page with the new value;
//    - a value of CALCULÉE from the state is not PAS a state: it is a
//      simple variable, recalculated each time the component is called;
//    - useMem    o : the same thing, but the calculation is only redone if its
//                    dependencies have changed (useful for a computationally expensive calculation);
//    - useEffec  t : code executed when the display is updated, when certain
//                    values have changed: a "side effect" (affecting something other
//                    thing other than the page: in this case, the tab title).
// =========================================================================

import { useEffect, useMemo, useState } from 'react';

// a formatter for euro amounts, in the French style (1,234.50 €)
const euros = new Intl.NumberFormat('fr-FR', { style: 'currency', currency: 'EUR' });

export default function App() {
  const [libelle, setLibelle] = useState('Stylo');
  const [prixHT, setPrixHT] = useState(2.5);
  const [quantite, setQuantite] = useState(1);
  const [tauxTVA, setTauxTVA] = useState(20); // as a percentage

   // values from CALCULÉES: simple constants, recalculated with each
   // call to the component. They are always up to date, with nothing to synchronize
  const totalHT = quantite * prixHT;
  const montantTVA = (totalHT * tauxTVA) / 100;
   // useMemo (calculation, [dépendances]): same result, but React only recalculates
   // recalculation unless one of the dependencies has changed since the previous call
  const totalTTC = useMemo(() => totalHT + montantTVA, [totalHT, montantTVA]);

   // the history of the totals TTC: it depends on the values SUCCESSIVES of the
   // total. The old total is stored in the report; if it differs from the
   // new one, the report is adjusted PENDANT when the component is called (React
   // then immediately repeats the call before displaying anything)
  const [historique, setHistorique] = useState<{ id: number; texte: string }[]>([]);
  const [ancienTTC, setAncienTTC] = useState(totalTTC);
  if (totalTTC !== ancienTTC) {
    setAncienTTC(totalTTC);
    const texte = `${euros.format(ancienTTC)} -> ${euros.format(totalTTC)}`;
     // Each row has a unique number (id) that identifies it (see key, example 03)
    setHistorique((h) => [{ id: (h[0]?.id ?? 0) + 1, texte }, ...h].slice(0, 5));
  }

   // useEffect (effect, [dépendances]): the effect is executed first
   // display, then after each display where totalTTC has changed
  useEffect(() => {
    document.title = `TTC : ${euros.format(totalTTC)}`;
  }, [totalTTC]);

   // changing the item resets the quantity to 1: this is done in the
   // event handler, where we know POURQUOI the status changes
  function changerLibelle(nouveau: string) {
    setLibelle(nouveau);
    setQuantite(1);
  }

  function reinitialiser() {
    setLibelle('Stylo');
    setPrixHT(2.5);
    setQuantite(1);
    setTauxTVA(20);
  }

  return (
    <div className="container py-4">
      <h1 className="h3 mb-3">Exemple 02 - l'état et les valeurs calculées</h1>
      <div className="row g-4">
        <div className="col-md-6">
          <div className="card p-3">
            {/* value + onChange: the field displays the status, and each keystroke changes it (see example 05) */}
            <label className="form-label" htmlFor="libelle">Article</label>
            <input id="libelle" className="form-control mb-2" value={libelle}
                   onChange={(e) => changerLibelle(e.target.value)} />
            <label className="form-label" htmlFor="prix">Prix unitaire HT</label>
            {/* valueAsNumber: the field’s value, as a number (NaN if empty) */}
            <input id="prix" type="number" step="0.1" min="0" className="form-control mb-2" value={prixHT}
                   onChange={(e) => setPrixHT(e.target.valueAsNumber || 0)} />
            <label className="form-label" htmlFor="quantite">Quantité (revient à 1 si l'article change)</label>
            <input id="quantite" type="number" min="0" className="form-control mb-2" value={quantite}
                   onChange={(e) => setQuantite(e.target.valueAsNumber || 0)} />
            <label className="form-label" htmlFor="tva">Taux de TVA</label>
            {/* the value of a <select> is a CHAÎNE: Number(...) converts it */}
            <select id="tva" className="form-select mb-3" value={tauxTVA}
                    onChange={(e) => setTauxTVA(Number(e.target.value))}>
              <option value={5.5}>5,5 %</option>
              <option value={10}>10 %</option>
              <option value={20}>20 %</option>
            </select>
            <button className="btn btn-outline-secondary" onClick={reinitialiser}>Réinitialiser</button>
          </div>
        </div>
        <div className="col-md-6">
          <div className="card p-3">
            <h2 className="h5">{quantite} x {libelle}</h2>
            <table className="table mb-0">
              <tbody>
                <tr><td>Total HT</td><td className="text-end">{euros.format(totalHT)}</td></tr>
                <tr><td>TVA ({tauxTVA} %)</td><td className="text-end">{euros.format(montantTVA)}</td></tr>
                <tr className="fw-bold"><td>Total TTC</td><td className="text-end">{euros.format(totalTTC)}</td></tr>
              </tbody>
            </table>
          </div>
          <div className="card p-3 mt-3">
            <h2 className="h6">Historique du total TTC</h2>
            <ul className="mb-0">
              {historique.map((ligne) => (
                <li key={ligne.id}>{ligne.texte}</li>
              ))}
            </ul>
          </div>
        </div>
      </div>
    </div>

    The difference between [Angular] and [Vue.js] is profound. In these frameworks, the data is reactive (signals, [ref]): the framework knows which data a section of the page uses, and updates only that section. [React] takes a simpler approach: every time the state changes, it re-runs the entire component function (and those of its children), then compares the result to the previous one. // 06-use-reducer/src/App.tsx - a form, an object, a reducer
// -------------------------------------------------------------------------
// Example 05 had one useState per field. For a real form, we
// prefer a UN object that contains all the values, the MODÈLE, and we
// describe these changes in a single function, RÉDUCTEUR:
//
//      reducer(etatActuel, action) -> nouvelEtat
//
// useReducer(reducer, etatInitial) returns the state and a function
// dispatch(action): the component no longer modifies the state; it DÉCRIT what
// what happened ({ type: 'update', field: 'name', value: 'Dupont' }), and the
// reducer derives the new state from this. All the logic is in one place,
// and it’s a PURE function (same input, same output, no side
// edge effects), which is easy to test.
//
// The state also tracks, for each field, whether it has been “touched” (exited at
// at least once); from this, we can determine whether it has been “modified” (its value differs from
// the initial value) and whether it is disabled.

import { useReducer, type ChangeEvent } from 'react';

// the format of the form data
interface Profil {
  titre: string;
  prenom: string;
  nom: string;
  age: number;
  newsletter: boolean;
  email: string;
  pays: string;
  naissance: string;
  presentation: string;
}

const DEPART: Profil = {
  titre: 'Mme',
  prenom: 'Marie',
  nom: 'PELISSIER',
  age: 30,
  newsletter: false,
  email: '',
  pays: 'fr',
  naissance: '',
  presentation: '',
};

type Champ = keyof Profil; // 'title' | 'first_name' | ...: the name of a field

interface Etat {
  valeurs: Profil;
  touches: Partial<Record<Champ, boolean>>; // fields already exited
}

// possible actions: a "discriminated union" based on the type property
type Action =
  | { type: 'modifier'; champ: Champ; valeur: Profil[Champ] }
  | { type: 'quitter'; champ: Champ }
  | { type: 'majuscules' }
  | { type: 'reinitialiser' };

// the reducer: it never returns the received report; it returns a new one
function reducteur(etat: Etat, action: Action): Etat {
  switch (action.type) {
    case 'modifier':
      return { ...etat, valeurs: { ...etat.valeurs, [action.champ]: action.valeur } };
    case 'quitter':
      return { ...etat, touches: { ...etat.touches, [action.champ]: true } };
    case 'majuscules': {
      const { prenom, nom } = etat.valeurs;
      return { ...etat, valeurs: { ...etat.valeurs, prenom: prenom.toUpperCase(), nom: nom.toUpperCase() } };
    }
    case 'reinitialiser':
      return { valeurs: { ...DEPART }, touches: {} };
  }
}

const CHAMPS_SUIVIS: Champ[] = ['prenom', 'nom', 'age', 'email'];

  }

  return (
    <div className="container py-4">
      <h1 className="h3 mb-3">Exemple 06 - un formulaire, un objet, un réducteur</h1>

// 07-validation/src/App.tsx - validate a form in the browser
// -------------------------------------------------------------------------
// React has no built-in validation system: it's just regular code. A
// function PURE, validate(account), returns field errors; it
// call it every time the form is displayed, so the errors are always up to date.
// They aren’t displayed right away (an empty form covered in red is
// unpleasant): they are shown only for a field that has already been exited (“touched”). The submission
// marks all fields as touched, then takes action only if the form
// is valid.
//
// Same rules as the RdvMedecins server ("Create an Account" form).
// ATTENTION: This validation is simply a CONFORT for the user (an
// an immediate response, without any back-and-forth with the server). It does not protect
// anything: anyone can send the server a request fabricated by
// (curl...). The server therefore revalidates TOUJOURS everything (see example 25).
// =========================================================================

import { useState, type FormEvent } from 'react';

interface Compte {
  titre: string;
  prenom: string;
  nom: string;
  login: string;
  password: string;
  confirmation: string;
}

type Champ = keyof Compte;
// at most one error message per field
type Erreurs = Partial<Record<Champ, string>>;

const TITRES = ['Mr', 'Mme', 'Melle'];
const FORMAT_LOGIN = /^[a-z0-9._-]{1,30}$/;
const MOT_DE_PASSE_MIN = 6;

// the rules: a pure function that depends only on its arguments
function valider(c: Compte): Erreurs {
  const erreurs: Erreurs = {};
  if (!TITRES.includes(c.titre)) erreurs.titre = 'Choisissez un titre dans la liste.';
  if (c.prenom.trim() === '') erreurs.prenom = 'Le prénom est obligatoire.';
  else if (c.prenom.length > 20) erreurs.prenom = 'Le prénom ne doit pas dépasser 20 caractères.';
  if (c.nom.trim() === '') erreurs.nom = 'Le nom est obligatoire.';
  else if (c.nom.length > 20) erreurs.nom = 'Le nom ne doit pas dépasser 20 caractères.';
  if (c.login === '') erreurs.login = 'Le login est obligatoire.';
  else if (!FORMAT_LOGIN.test(c.login)) erreurs.login = 'Lettres minuscules non accentuées, chiffres, point, tiret, souligné.';
  if (c.password === '') erreurs.password = 'Le mot de passe est obligatoire.';
  else if (c.password.length < MOT_DE_PASSE_MIN) erreurs.password = `Au moins ${MOT_DE_PASSE_MIN} caractères.`;
   // a rule that compares DEUX fields
  if (c.confirmation !== c.password) erreurs.confirmation = 'La confirmation ne correspond pas au mot de passe.';
  return erreurs;
}

const VIDE: Compte = { titre: 'Mr', prenom: '', nom: '', login: '', password: '', confirmation: '' };
const TOUS: Record<Champ, boolean> = { titre: true, prenom: true, nom: true, login: true, password: true, confirmation: true };

export default function App() {
  const [compte, setCompte] = useState<Compte>(VIDE);
  const [touches, setTouches] = useState<Partial<Record<Champ, boolean>>>({});
  const [envoiTente, setEnvoiTente] = useState(false);
  const [succes, setSucces] = useState(false);

   // Errors: recalculated with each display, never saved in the report
  const erreurs = valider(compte);
  const nbErreurs = Object.keys(erreurs).length;

   // A field’s Bootstrap class: red (is-invalid) or green (is-valid),
   // only once it has been selected
  function classe(champ: Champ, base = 'form-control'): string {
    if (!touches[champ]) return base;
    return `${base} ${erreurs[champ] ? 'is-invalid' : 'is-valid'}`;
  }

   // attributes common to all fields
  function lier(champ: Champ) {
    return {
      id: champ,
      value: compte[champ],
      onChange: (e: { target: { value: string } }) => {
        setCompte((c) => ({ ...c, [champ]: e.target.value }));
        setSucces(false);
      },
      onBlur: () => setTouches((t) => ({ ...t, [champ]: true })),
    };
  }

  function envoyer(evenement: FormEvent<HTMLFormElement>) {
    evenement.preventDefault();
    setEnvoiTente(true);
    setTouches(TOUS); // all errors appear
     // here, the form would be submitted to the server (see Example 25)
    setSucces(nbErreurs === 0);
  }

// -------------------------------------------------------------------------
// React does not have its own formatting mechanism (Angular has its
// "pipes," Vue.js its functions): in JSX, we call FONCTIONS
// ordinary functions, organized into a module and testable separately:
//      {schedule(time          slot)} -> "8:00 - 8:20"
//      {time(time slot, 'start')} -> "8:00"
// For dates, numbers, and amounts, the browser provides the API
// Intl, which understands the conventions of each language.

export interface Creneau {
  hdebut: number;
  mdebut: number;
  hfin: number;
  mfin: number;
}

// "8:05" rather than "8:5"
function heure(h: number, m: number): string {
  return `${h}:${String(m).padStart(2, '0')}`;
}

export function horaire(c: Creneau, partie: 'tout' | 'debut' | 'fin' = 'tout'): string {
  switch (partie) {
    case 'debut':
      return heure(c.hdebut, c.mdebut);
    case 'fin':
      return heure(c.hfin, c.mfin);
    default:
      return `${heure(c.hdebut, c.mdebut)} - ${heure(c.hfin, c.mfin)}`;
  }
}

// "marie pelissier" -> "Marie Pelissier"
export function majusculeInitiale(texte: string): string {
  return texte.replace(/\p{L}+/gu, (mot) => mot[0].toUpperCase() + mot.slice(1).toLowerCase());
}

// "Marie PELISSIER" -> "MP"
export function initiales(nomComplet: string): string {
  return nomComplet
    .split(/\s+/)
    .filter((mot) => mot !== '')
    .map((mot) => mot[0].toUpperCase())
    .join('');
}

// "2026-10-05" (the format of API RdvMedecins) -> a date, at midnight UTC
export function versDate(jour: string): Date {
  return new Date(`${jour}T00:00:00Z`);
}
