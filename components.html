// 10-components/src/CarteMedecin.tsx - a component ENFANT
// -------------------------------------------------------------------------
// It receives its data from its parent via PROPS (properties): the
// attributes written by the parent to <CarteMedecin ... />, grouped by
// React into a UN object, the first parameter of the function. Their type is
// checked at compile time: <CarteMedecin doctor={42} /> would be rejected.
//
// Golden rule: a component does not MODIFIE JAMAIS its props. The data
// flows from the parent to the child; to request a change,
// the child calls a function passed to it by the parent (see Example 11).
// =========================================================================

import type { Medecin } from './medecin';
// styles: an object { initials: '_initiales_x7f3a_1' } - the actual name of each class
import styles from './CarteMedecin.module.css';

// the type of props: what the parent PEUT (?) or DOIT provides
interface Props {
  medecin: Medecin;
  rang: number;
  compact?: boolean; // optional
}

// the object is "unpacked" from the props; compact = false: its default value
export default function CarteMedecin({ medecin, rang, compact = false }: Props) {
   // a value calculated from the props: recalculated if the parent changes them
  const initiales = `${medecin.prenom[0]}${medecin.nom[0]}`;

  return (
    <div className={`card h-100 ${medecin.disponible ? 'border-success' : ''}`}>
      <div className="card-body d-flex gap-3 align-items-center">
        <div className={styles.initiales}>{initiales}</div>
        <div>
          <h2 className="h6 mb-1">{rang}. {medecin.titre} {medecin.prenom} {medecin.nom}</h2>
          {!compact && <p className="mb-0 text-muted small">{medecin.specialite}</p>}
        </div>
      </div>
    </div>
  );
}
    • line 14: a type import only;
    • line 16: an import of the CSS module (see below);
    • lines 19–23: the component’s prop types—the attributes that the parent writes to [<CarteMedecin ... />]. [compact?] is optional;
    • line 26: [React] combines the props into a UN object, which it passes as the first parameter to the function. It is immediately “unpacked”: [{ medecin, rang, compact = false }] yields three variables, with [compact] being equal to [false] if the parent does not provide it. The type of the props is checked at compile time: [<CarteMedecin medecin={42} />] would be rejected, as would the omission of [rang];
    • line 28: a value calculated from the props: if the parent changes the props, the component is re-invoked and the value is recalculated;
    • line 33: the class CSS, read from the module CSS;
    • line 36: the specialty is displayed only in normal mode.

    Golden rule: a component never modifies its own props. Data flows from the parent to the child; to request a change, the child calls a function passed to it by the parent (see Example 11).

    { id: 2, titre: 'Mr', prenom: 'Paul', nom: 'TATOU', specialite: 'pédiatre', disponible: false },
    { id: 3, titre: 'Melle', prenom: 'Sophie', nom: 'DUPONT', specialite: 'généraliste', disponible: true },
  ]);

   // children, along with their new props
  function inverserDisponibilite() {
    setMedecins((liste) => liste.map((m) => ({ ...m, disponible: !m.disponible })));
  }

  return (
    <div className="container py-4">
      <h1 className="h3 mb-3">Exemple 10 - les composants et leurs props</h1>
      <div className="d-flex gap-3 mb-3">
        <label className="form-check-label">
          <input type="checkbox" className="form-check-input" checked={compact}
                 onChange={(e) => setCompact(e.target.checked)} /> affichage compact
        </label>
        <button className="btn btn-sm btn-outline-primary" onClick={inverserDisponibilite}>Inverser les disponibilités</button>
      </div>
      <div className="row g-3">
        {medecins.map((m, i) => (
          <div key={m.id} className="col-md-4">
            {/* doctor={m} passes the OBJET m (an expression);
                medecin="m", sans accolades, transmettrait la CHAÎNE "m" */}

        <CarteMedecin medecin={medecins[0]} rang={1} compact />
      </div>
    </div>
  );
}

    • lines 25–27: the parent modifies SES data. [React] re-renders it, then re-renders the children with their new props: the maps update;
    • line 44: the component is used as a new tag. [medecin={m}] passes OBJET to [m] (an expression); [medecin="m"], without curly braces, would pass the CHAÎNE “m”;
    • line 51: an attribute without a value ([compact]) is equal to [true].

    // =========================================================================
// 11-retrieve-the-information/src/Compteur.tsx
// -------------------------------------------------------------------------
// A child that PRÉVIENT its parent: the parent passes it FONCTIONS
// in their props (by convention, their names begin with "on"), and the child
// calls them when something happens:
//      <Counter label="Coffees" max={3} onIncremente={...} onLimite={...} />
// =========================================================================

import { useState } from 'react';

interface Props {
  libelle: string;
  max: number;
  onIncremente: (valeur: number) => void; // with the new value
  onLimite: () => void; // with no data
}

export default function Compteur({ libelle, max, onIncremente, onLimite }: Props) {
   // the status PROPRE for this counter: each <Counter> has its own
  const [valeur, setValeur] = useState(0);

  function plus() {
    if (valeur >= max) {
      onLimite();
      return;
    }
    setValeur(valeur + 1);
     // value has not yet been updated (it will be on the next call to the
     // component): the value is sent as +1
    onIncremente(valeur + 1);
  }

  return (
    <div className="d-flex align-items-center gap-2 mb-2">
      <span className="w-25">{libelle}</span>
      <button className="btn btn-sm btn-outline-primary" onClick={plus}>+1</button>
      <span className="badge text-bg-secondary">{valeur} / {max}</span>
    </div>
  );
}
    • Lines 12–17: Two props are functions: [onIncremente] receives the new value, while [onLimite] receives nothing. By convention, their names begin with “on,” such as [onClick];
    • Line 21: The state PROPRE for this counter: each [<Compteur>] on the page has its own. The state is attached to the component’s position in the page tree;
    • lines 23–32: the parent is notified. Note: [setValeur(valeur + 1)] does not change the variable [valeur] (a constant set to CET when the component is called): it will be [valeur + 1] at the time of the PROCHAIN call. Therefore, [valeur + 1] is passed to the parent.

// 11-report-the-information/src/ChampTexte.tsx - a CONTRÔLÉ component
// -------------------------------------------------------------------------
// A "custom" form field (label + field + tooltip + character counter
// ) that is used as a controlled <input> (see example 05):
//      <ChampTexte value={name} onChange={setNom} label="Name" max={20} />
// The component has no status: the value belongs to the parent, which passes it to the component
// (value) and notifies it of every keystroke (onChange). It is the

import { useId } from 'react';

interface Props {
  valeur: string;
  onChange: (valeur: string) => void;
  libelle: string;
  max: number;
  aide?: string;
}

export default function ChampTexte({ valeur, onChange, libelle, max, aide = '' }: Props) {
   // useId: an identifier UNIQUE, which is different for each <ChampTexte> on the
   // page, to link <label htmlFor> and <input id>
  const id = useId();

  return (
    <div className="mb-3">
      <label className="form-label" htmlFor={id}>{libelle}</label>
      <input id={id} className={`form-control ${valeur.length === max ? 'is-invalid' : ''}`} value={valeur}
             maxLength={max} onChange={(e) => onChange(e.target.value)} />
      <div className="form-text">{aide} ({valeur.length} / {max})</div>
    </div>
  );
}

    A “custom” form field (label, field, tooltip, character counter) that is used as a controlled [<input>] (see Example 05): the parent passes it the value ([valeur]) and a function called with each keystroke ([onChange]).

    • lines 14–20: the props; [aide] is optional;
    • line 25: [useId()] provides a unique identifier, different for each [<ChampTexte>] on the page, to link [<label htmlFor>] and [<input id>];
    • line 31: Each keystroke calls the parent’s function with the new text.

    The component has no state: the value belongs to the parent, which determines the new value. This is called “propagating state”: data shared by multiple components is stored in their closest common ancestor.

// -------------------------------------------------------------------------
// Data flows down (props); information flows back up via

// 12-composition/src/Carte.tsx - a reusable GABARIT
// -------------------------------------------------------------------------
// Props pass DONNÉES... but also CONTENU: JSX is
// a value like any other (of type ReactNode), which can be passed to a
// component, which places it wherever it wants:
//    - children: the special property that receives whatever the parent writes ENTRE
//                 <Map> and </Map>—the “main” content;
//    - header, footer: other sections, provided by the parent as
//                 the form of props; the header has default content.
// The card defines the frame (border, header, footer); the parent provides
// content inside it. This is the client-side equivalent of the template layout.hbs
// and its {{{body}}} in the NestJS course.
// =========================================================================

import type { ReactNode } from 'react';

interface Props {
  couleur?: 'primary' | 'success' | 'danger';
  entete?: ReactNode;
  pied?: ReactNode;
  children: ReactNode;
}

export default function Carte({ couleur, entete = 'Sans titre', pied, children }: Props) {
  return (
    <div className={`card mb-3 ${couleur ? `border-${couleur}` : ''}`}>
      <div className={`card-header ${couleur ? `text-bg-${couleur}` : ''}`}>{entete}</div>
      <div className="card-body">{children}</div>
      {/* no footer provided: no footer at all */}
      {pied && <div className="card-footer text-muted small">{pied}</div>}
    </div>
  );
}
    • lines 18–23: [entete], [pied], and [children] are [ReactNode]: JSX, text, a number, or nothing;
    • line 25: default content;
    • line 29: [children] is a special property: it receives whatever the parent writes in ENTRE, [<Carte>], and [</Carte>]—the “main” content;
    • line 31: no footer provided—no footer at all.

    The template defines the framework (border, header, footer); the parent provides the content inside it: this is equivalent to the content projection of [Angular] ([<ng-content>]) and the “slots” of [Vue.js], and, on the server side, the template [layout.hbs] and its [{{{body}}}] within the course [NestJS].

// 12-composition/src/Tableau.tsx - a table GÉNÉRIQUE
// -------------------------------------------------------------------------
// Array<T> is a generic component: it displays rows of
// of any type T. It knows how to iterate through the rows, but not how
// display UNE a row: the parent tells it how by passing it a
// FONCTION, renderLigne(row, index), which returns the JSX of the cells.
// This is what’s called a “render prop”: the content provided by the
// parent uses data belonging to the child (the row, its
// rank).
// =========================================================================

import type { ReactNode } from 'react';

interface Props<T> {
  lignes: T[];
  colonnes: string[];
  renderLigne: (ligne: T, index: number) => ReactNode;
  vide?: ReactNode; // displayed when there are no rows
}

// <T extends { id: number }>: each row must have an id (the key)
export default function Tableau<T extends { id: number }>({ lignes, colonnes, renderLigne, vide = 'Aucune donnée.' }: Props<T>) {
  return (
    <table className="table table-sm align-middle mb-0">
      <thead>
        <tr>
          {colonnes.map((c) => <th key={c}>{c}</th>)}
        </tr>
      </thead>
      <tbody>
        {lignes.length === 0 ? (
          <tr>
            <td className="text-muted" colSpan={colonnes.length}>{vide}</td>
          </tr>
        ) : (
          lignes.map((l, i) => <tr key={l.id}>{renderLigne(l, i)}</tr>)
        )}
      </tbody>
    </table>
  );
}
    • lines 15–20: a generic type: the props of an array of lines of type [T], whatever they may be;
    • line 18: a prop that is a function returning JSX: the array knows how to iterate over the rows, but not how to display UNE per row; the parent tells it how. This process is called a “render prop”;
    • line 23: a generic component: [T] is inferred at runtime from the type of the rows. Each row must have a [id], which serves as a key;
    • line 37: For each line, the array calls the parent’s function, passing it data that belongs to it (the line, its index).

// The life cycle of a component:
//    1. It is AFFICHÉ (mounted): React calls the function and places its
//       result on the page; then it executes its effects (useEffect);
//    2. it is MIS to JOUR, as many times as necessary (its state or

import { useEffect, useEffectEvent, useState } from 'react';

interface Props {
  onJournal: (message: string) => void;
}

export default function Horloge({ onJournal }: Props) {

   // useEffectEvent: a function called PAR by an effect, which always reads
   // the most recent props, without being a dependency of the effect. Without
   // it, we would have to include onJournal in the dependencies [...] below,
   // and, since the parent passes a NOUVELLE function on every render, the
   // timer would be stopped and restarted every second
  const journaliser = useEffectEvent((message: string) => onJournal(message));

   // [] : no dependencies. The effect runs UNE times, after the first
   // display; the function it returns runs UNE times when the component is removed
  useEffect(() => {
    journaliser('Horloge : effet - démarrage de la minuterie');
    const minuterie = setInterval(() => setHeure(new Date().toLocaleTimeString()), 1000);
     // the NETTOYAGE
    return () => {
      clearInterval(minuterie);

    };
  }, []);

  lignes: string[];

// The provider: it contains the state and passes it down to its descendants
export function JournalFournisseur({ children }: { children: ReactNode }) {
  const [lignes, setLignes] = useState<string[]>([]);

   // useCallback: the MÊME function varies from one rendering to the next (and is not a
   // new one each time), so that the components that receive it
   // don’t appear to have changed
  const tracer = useCallback((message: string) => {
    setLignes((l) => [`${new Date().toLocaleTimeString()} ${message}`, ...l].slice(0, 6));

  return <JournalContexte value={valeur}>{children}</JournalContexte>;

  return journal;

// 16-confirmation/src/confirmation/confirmation.tsx
// -------------------------------------------------------------------------
// “Are you sure?”: window.confirm() exists, but we can neither
// style it nor translate its buttons. We replace it with a window
// Bootstrap dialog, controlled by a context (see Example 14).
//
// Tip: confirm(message) returns a PROMESSE, which evaluates to true (Yes)
// or false (No) when the user responds. The calling code is written
// just as simply as with window.confirm:
//
//      const confirm = useConfirmation();
//      if (await confirm('Delete this doctor?')) { ... }
//
// The provider displays the window itself: simply place it
// time around the entire application (see main.tsx).
// =========================================================================

import { createContext, use, useCallback, useRef, useState, type ReactNode } from 'react';
import FenetreConfirmation from './FenetreConfirmation';

type Confirmer = (message: string) => Promise<boolean>;

const ConfirmationContexte = createContext<Confirmer | null>(null);

export function ConfirmationFournisseur({ children }: { children: ReactNode }) {
  const [message, setMessage] = useState<string | null>(null); // null: window closed
   // the function that will resolve the current promise: it is not displayed,
   // so it is a reference and not a state
  const repondre = useRef<((oui: boolean) => void) | null>(null);

  const confirmer = useCallback<Confirmer>((texte) => {
    repondre.current?.(false); // Is there still an open question? It is set to "no"
    setMessage(texte);
    return new Promise<boolean>((resolve) => {
      repondre.current = resolve;
    });
  }, []);

  function reponse(oui: boolean) {
    setMessage(null);
    const r = repondre.current;
    repondre.current = null;
    r?.(oui);
  }

  return (
    <ConfirmationContexte value={confirmer}>
      {children}
      {message !== null && <FenetreConfirmation message={message} onReponse={reponse} />}
    </ConfirmationContexte>
  );
}

export function useConfirmation(): Confirmer {
  const confirmer = use(ConfirmationContexte);
  if (confirmer === null) {
    throw new Error('useConfirmation doit être utilisé dans un <ConfirmationFournisseur>');
  }
  return confirmer;
}

    Tip: [confirmer(message)] returns a promise, which resolves to [true] (Yes) or [false] (No) when the user responds. The calling code is then just as simple as with [window.confirm]: [if (await confirmer('Supprimer ce médecin ?')) {...}].

// The modal window:
//    - Same Bootstrap classes as a real modal (.modal, .modal-backdrop),

//      Bootstrap;
//    - createPortal: the window is placed at the end of <body>, not where
//      the component is written. It thus bypasses the styles of its

import { useEffect } from 'react';
import { createPortal } from 'react-dom';

interface Props {
  message: string;
  onReponse: (oui: boolean) => void;
}

export default function FenetreConfirmation({ message, onReponse }: Props) {
   // Esc, anywhere on the page: a listener attached to the DOCUMENT
   // upon opening, removed upon closing (clearing the effect)
  useEffect(() => {
    const clavier = (e: KeyboardEvent) => {
      if (e.key === 'Escape') onReponse(false);
    };
    document.addEventListener('keydown', clavier);
    return () => document.removeEventListener('keydown', clavier);
  }, [onReponse]);

  return createPortal(
    <>
      <div className="modal-backdrop fade show"></div>
      {/* A click on the background (not inside the window) counts as “no” */}
      <div className="modal fade show d-block" tabIndex={-1} role="alertdialog" aria-modal="true"
           onClick={(e) => e.target === e.currentTarget && onReponse(false)}>
        <div className="modal-dialog modal-dialog-centered">
          <div className="modal-content">
            <div className="modal-header">
              <h5 className="modal-title">Confirmation</h5>
              <button type="button" className="btn-close" aria-label="Fermer" onClick={() => onReponse(false)}></button>
            </div>
            <div className="modal-body">
              <p className="mb-0">{message}</p>
            </div>
            <div className="modal-footer">
              {/* autoFocus: React gives focus to this button when it appears */}
              <button type="button" className="btn btn-secondary" autoFocus onClick={() => onReponse(false)}>Non, revenir</button>
              <button type="button" className="btn btn-danger" onClick={() => onReponse(true)}>Oui, confirmer</button>
            </div>
          </div>
        </div>
      </div>
    </>,
    document.body,
  );
}

    • line 35: [createPortal(jsx, élément)] displays JSX somewhere other than where the component is written: here, at the end of [<body>]. The window thus bypasses the styles of its ancestors ([overflow], [z-index]...), while remaining an ordinary [React] component (props, events);
